alembic.ini:
```
# Alembic: миграции схемы БД. Запуск из папки backend/:
#   alembic upgrade head
#   alembic revision --autogenerate -m "что изменили"
# Адрес БД берётся из DATABASE_URL (.env), а не из этого файла.

[alembic]
script_location = migrations
prepend_sys_path = .

[loggers]
keys = root,sqlalchemy,alembic

[handlers]
keys = console

[formatters]
keys = generic

[logger_root]
level = WARN
handlers = console
qualname =

[logger_sqlalchemy]
level = WARN
handlers =
qualname = sqlalchemy.engine

[logger_alembic]
level = INFO
handlers =
qualname = alembic

[handler_console]
class = StreamHandler
args = (sys.stderr,)
level = NOTSET
formatter = generic

[formatter_generic]
format = %(levelname)-5.5s [%(name)s] %(message)s
datefmt = %H:%M:%S

```

Dockerfile:
```
FROM python:3.12-slim

WORKDIR /app

ENV PYTHONUNBUFFERED=1 \
    PYTHONDONTWRITEBYTECODE=1 \
    PIP_NO_CACHE_DIR=1

COPY requirements.txt .
RUN pip install -r requirements.txt

COPY . .

EXPOSE 5000

# entrypoint.sh доведёт схему БД до актуальной и создаст администратора из .env
ENTRYPOINT ["sh", "./entrypoint.sh"]
CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "5000"]

```

entrypoint.sh:
```
#!/bin/sh
set -e

# Схема БД: создаём базу и таблицы, если их нет.
alembic upgrade head

# Проверяем, что схема совпадает с моделями. create_all не меняет уже
# существующие таблицы, поэтому после правки модели приложение упало бы
# позже и не там, где ошибку ищут. Лучше не подняться вовсе и сказать почему.
python -m scripts.check_schema

# Администратор из .env + тестовые данные (идемпотентно, дубли не создаются).
python -m scripts.seed_all

exec "$@"

```

"""Пароли, JWT и проверка прав.

Токен передаётся в заголовке `Authorization: Bearer <token>`.
Права проверяются на сервере в каждом эндпоинте через зависимости
CurrentUser / AdminUser, поэтому ограничения действуют и в API, а не только в интерфейсе.
"""

import uuid
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from typing import Annotated

import bcrypt
import jwt
from fastapi import Depends, Request
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from jwt.exceptions import ExpiredSignatureError, InvalidTokenError
from sqlalchemy.orm import Session

from app.config import settings
from app.db import get_db
from app.errors import ApiError
from app.logging_config import get_logger
from app.models import RevokedToken, User

logger = get_logger(__name__)

# auto_error=False: без токена отдаём 401 (а не 403, как по умолчанию в FastAPI)
bearer_scheme = HTTPBearer(auto_error=False, description="JWT из POST /api/v1/auth/login")


def hash_password(password: str) -> str:
    """Хеш пароля с bcrypt"""
    return bcrypt.hashpw(password.encode()[:72], bcrypt.gensalt()).decode()


def verify_password(plain_password: str, hashed_password: str) -> bool:
    """Проверяет пароль с хешем"""
    try:
        return bcrypt.checkpw(plain_password.encode()[:72], hashed_password.encode())
    except ValueError:
        return False


def create_access_token(user_id: int, expires_delta: timedelta | None = None) -> tuple[str, datetime]:
    """Создаём JWT-токен. Возвращает (токен, момент истечения в UTC)."""
    expire = datetime.now(timezone.utc) + (
        expires_delta or timedelta(minutes=settings.access_token_expire_minutes)
    )
    payload = {
        "sub": str(user_id),  # PyJWT требует строку в sub
        "jti": uuid.uuid4().hex,  # id токена — нужен для logout
        "exp": expire,  # стандартное поле: PyJWT сам проверяет истечение
    }
    return jwt.encode(payload, settings.jwt_secret_key, algorithm=settings.jwt_algorithm), expire


def decode_access_token(token: str) -> dict:
    """Декодируем JWT токен"""
    try:
        return jwt.decode(
            token,
            settings.jwt_secret_key,
            algorithms=[settings.jwt_algorithm],
            options={"require": ["exp", "sub", "jti"]},
        )
    except ExpiredSignatureError:
        raise ApiError(401, "Срок действия токена истёк")
    except InvalidTokenError:
        raise ApiError(401, "Недействительный токен")


@dataclass
class AuthContext:
    user: User
    token_payload: dict


def get_auth(
    credentials: Annotated[HTTPAuthorizationCredentials | None, Depends(bearer_scheme)],
    db: Annotated[Session, Depends(get_db)],
) -> AuthContext:
    if credentials is None or credentials.scheme.lower() != "bearer":
        raise ApiError(401, "Требуется авторизация")

    payload = decode_access_token(credentials.credentials)
    if db.get(RevokedToken, payload["jti"]):
        raise ApiError(401, "Сессия завершена, войдите снова")

    user = db.get(User, int(payload["sub"]))
    if user is None:
        raise ApiError(401, "Пользователь не найден")
    return AuthContext(user=user, token_payload=payload)


def get_current_user(auth: Annotated[AuthContext, Depends(get_auth)]) -> User:
    """Любой авторизованный пользователь."""
    return auth.user

        raise ApiError(403, "Действие доступно только администратору")
    return user

# Явные имена constraint'ов — без них Alembic в batch-режиме SQLite
# не сможет удалять/менять ограничения в следующих миграциях.
NAMING_CONVENTION = {
    "ix": "ix_%(column_0_label)s",
    "uq": "uq_%(table_name)s_%(column_0_name)s",
    "ck": "ck_%(table_name)s_%(constraint_name)s",
    "fk": "fk_%(table_name)s_%(column_0_name)s_%(referred_table_name)s",
    "pk": "pk_%(table_name)s",
}


class Base(DeclarativeBase):
    metadata = MetaData(naming_convention=NAMING_CONVENTION)


def make_engine(url: str):
    parsed = make_url(url)
    connect_args = {}
    if parsed.get_backend_name() == "sqlite":
        # FastAPI выполняет sync-эндпоинты в пуле потоков
        connect_args["check_same_thread"] = False
        if parsed.database and parsed.database != ":memory:":
            Path(parsed.database).parent.mkdir(parents=True, exist_ok=True)

    engine = create_engine(url, connect_args=connect_args)

    if parsed.get_backend_name() == "sqlite":

        @event.listens_for(engine, "connect")
        def _sqlite_pragmas(dbapi_connection, connection_record):
            """Внешние ключи + WAL: бэкенд и worker пишут в один файл."""
            cursor = dbapi_connection.cursor()
            cursor.execute("PRAGMA foreign_keys=ON")
            cursor.execute("PRAGMA journal_mode=WAL")
            cursor.execute("PRAGMA busy_timeout=5000")
            cursor.close()

    return engine

def schema_drift(target_engine=None) -> tuple[list[str], list[str]]:
    """Сравнивает схему базы с моделями.

    Возвращает (недостающее, лишнее):
    - **недостающее** — в базе нет таблицы или колонки, которая есть в моделях.
      Приложение упадёт на первой же записи (no such column / NOT NULL);
    - **лишнее** — в базе есть то, чего в моделях нет. На работу не влияет,
      но значит, что база осталась от прежней версии схемы.

    Нужно потому, что create_all умеет только СОЗДАВАТЬ таблицы: изменили
    модель — база молча осталась старой, и ошибка всплывает позже и не там,
    где её искали.
    """
    # Модели регистрируются в Base.metadata при импорте пакета app.models.
    # Импорт внутри функции: app.models импортирует app.db — на уровне модуля
    # получился бы цикл, а здесь он безопасен.
    from app import models  # noqa: F401

    inspector = inspect(target_engine or engine)
    db_tables = set(inspector.get_table_names())
    meta_tables = set(Base.metadata.tables)

    missing: list[str] = []
    extra: list[str] = []

    for name in sorted(meta_tables - db_tables):
        missing.append(f"таблица {name}")
    for name in sorted(db_tables - meta_tables - {"alembic_version"}):
        extra.append(f"таблица {name}")

    for name in sorted(meta_tables & db_tables):
        db_columns = {c["name"] for c in inspector.get_columns(name)}
        meta_columns = set(Base.metadata.tables[name].columns.keys())
        missing.extend(f"{name}.{col}" for col in sorted(meta_columns - db_columns))
        extra.extend(f"{name}.{col}" for col in sorted(db_columns - meta_columns))

    return missing, extra

    {"code": 403, "status": "Forbidden", "message": "Действие доступно только администратору"}

from http import HTTPStatus

from fastapi.responses import JSONResponse
from starlette.exceptions import HTTPException as StarletteHTTPException

from app.logging_config import get_logger

logger = get_logger(__name__)


class ApiError(Exception):
    def __init__(self, code: int, message: str):
        self.code = code
        self.message = message


def _body(code: int, message: str) -> dict:
    # В Python 3.13 422 называется «Unprocessable Content»; оставляем прежнее название
    phrase = "Unprocessable Entity" if code == 422 else HTTPStatus(code).phrase
    return {"code": code, "status": phrase, "message": message}


def register_error_handlers(app: FastAPI) -> None:
    @app.exception_handler(ApiError)
    async def api_error(request: Request, exc: ApiError):
        return JSONResponse(_body(exc.code, exc.message), status_code=exc.code)

    @app.exception_handler(StarletteHTTPException)
    async def http_error(request: Request, exc: StarletteHTTPException):
        message = exc.detail if isinstance(exc.detail, str) else HTTPStatus(exc.status_code).phrase
        return JSONResponse(_body(exc.status_code, message), status_code=exc.status_code, headers=exc.headers)

    @app.exception_handler(RequestValidationError)
    async def validation_error(request: Request, exc: RequestValidationError):
        errors = exc.errors()
        message = errors[0].get("msg", "Некорректный запрос") if errors else "Некорректный запрос"
        # Свои ValueError в валидаторах pydantic оборачивает в «Value error, …»
        return JSONResponse(_body(422, message.removeprefix("Value error, ")), status_code=422)

    @app.exception_handler(Exception)
    async def unhandled(request: Request, exc: Exception):
        logger.exception(f"Необработанная ошибка: {request.method} {request.url.path}")
        return JSONResponse(_body(500, "Внутренняя ошибка сервера"), status_code=500)

"""Работа с датами в едином часовом поясе приложения."""

import calendar
from datetime import date, datetime, timezone
from zoneinfo import ZoneInfo

from app.config import settings


def app_tz() -> ZoneInfo:
    return ZoneInfo(settings.app_timezone)


# Для тестов: можно «заморозить» сегодняшнюю дату (timeutils.FROZEN_TODAY = date(...))
FROZEN_TODAY: date | None = None


def today() -> date:
    """«Сегодня» в часовом поясе приложения (не сервера!)."""
    if FROZEN_TODAY is not None:
        return FROZEN_TODAY
    return datetime.now(app_tz()).date()


def utcnow() -> datetime:
    """Текущий момент в UTC без tzinfo — так храним метки времени в БД."""
    return datetime.now(timezone.utc).replace(tzinfo=None)


def due_date_for(anchor_day: int, year: int, month: int) -> date:
    """Дата оплаты в месяце: если такого числа нет — последний день месяца."""
    last_day = calendar.monthrange(year, month)[1]
    return date(year, month, min(anchor_day, last_day))


def add_months(start: date, months: int, anchor_day: int | None = None) -> date:
    """Сдвиг на N месяцев, всегда от исходного числа (anchor_day).

    Важно: считаем от anchor_day, а не от предыдущей даты, иначе
    31.01 → 28.02 → 28.03 («уползание»). Здесь 31.01 → 28.02 → 31.03.
    """
    anchor = anchor_day or start.day
    total = start.year * 12 + (start.month - 1) + months
    year, month = divmod(total, 12)
    return due_date_for(anchor, year, month + 1)

"""Фоновый процесс напоминаний.

Запуск: python -m app.worker
Раз в REMINDER_CHECK_INTERVAL_SECONDS досоздаёт периоды и отправляет напоминания.
Должен работать ровно в одном экземпляре (отдельный контейнер `worker`).
"""

import signal
import time

from app.config import settings
from app.db import SessionLocal
from app.logging_config import get_logger, setup_logging
from app.services import telegram
from app.services.reminders import check_and_send

logger = get_logger("worker")

_running = True


def _stop(signum, frame):
    global _running
    logger.info(f"Получен сигнал {signum}, останавливаю worker")
    _running = False


def run_once() -> int:
    """Один проход проверки. Возвращает число отправленных напоминаний."""
    db = SessionLocal()
    try:
        return check_and_send(db)
    except Exception:
        logger.exception("Ошибка при проверке напоминаний")
        db.rollback()
        return 0
    finally:
        db.close()


def main():
    setup_logging(name="worker")
    signal.signal(signal.SIGTERM, _stop)
    signal.signal(signal.SIGINT, _stop)

    # Нижняя граница: при REMINDER_CHECK_INTERVAL_SECONDS=0 получился бы бесконечный цикл по БД
    interval = max(settings.reminder_check_interval_seconds, 5)
    logger.info(
        f"Worker запущен: проверка каждые {interval} с, напоминание за "
        f"{settings.reminder_days_before} дн., часовой пояс {settings.app_timezone}"
    )
    if not telegram.is_configured():
        logger.warning("Telegram не настроен — напоминания отправляться не будут")

    while _running:
        sent = run_once()
        if sent:
            logger.info(f"Отправлено напоминаний: {sent}")
        # Спим маленькими шагами, чтобы быстро реагировать на остановку контейнера
        for _ in range(interval):
            if not _running:
                break
            time.sleep(1)

    logger.info("Worker остановлен")


if __name__ == "__main__":
    main()

from app.db import Base
from app.timeutils import utcnow

STATUS_PAID = "paid"
STATUS_UNPAID = "unpaid"
STATUSES = (STATUS_PAID, STATUS_UNPAID)

# Вычисляемое состояние периода (в БД не хранится)
STATE_PENDING = "pending"  # не оплачено, срок не прошёл
STATE_OVERDUE = "overdue"  # не оплачено, срок прошёл
STATE_PAID = "paid"  # оплачено вовремя (в том числе в день срока)
STATE_PAID_LATE = "paid_late"  # оплачено после срока
STATES = (STATE_PENDING, STATE_OVERDUE, STATE_PAID, STATE_PAID_LATE)


class PaymentOccurrence(Base):
    """Конкретный период регулярного платежа: своя дата и своя отметка оплаты.

    Название, сумма и валюта — копия полей шаблона на момент создания периода.
    Шаблон нельзя изменить через API (в кейсе есть только создать и прекратить),
    поэтому данные не расходятся, а список периодов читается из одной таблицы
    без JOIN.
    """

    __tablename__ = "payment_occurrences"
    __table_args__ = (
        # Один период на дату — защита от дублей при параллельной генерации
        UniqueConstraint("recurring_payment_id", "due_date", name="uq_occurrence_payment_due_date"),
    )

    id: Mapped[int] = mapped_column(Integer, primary_key=True)
    recurring_payment_id: Mapped[int] = mapped_column(
        ForeignKey("recurring_payments.id", ondelete="CASCADE"), index=True
    )
    # Слепок шаблона: период читается сам по себе, без JOIN
    name: Mapped[str] = mapped_column(String(200))
    amount: Mapped[Decimal] = mapped_column(Numeric(14, 2))
    currency: Mapped[str] = mapped_column(String(3), index=True)
    due_date: Mapped[date] = mapped_column(Date, index=True)
    status: Mapped[str] = mapped_column(String(16), default=STATUS_UNPAID, index=True)
    # Фактическая дата оплаты; при отмене отметки очищается
    paid_at: Mapped[date | None] = mapped_column(Date)
    marked_by_id: Mapped[int | None] = mapped_column(ForeignKey("users.id", ondelete="SET NULL"))
    # Когда платёж прекратили: NULL → платёж активен
    payment_ended_at: Mapped[datetime | None] = mapped_column(DateTime)
    # Когда отправлено напоминание: NULL → ещё не напоминали (защита от дублей)
    reminder_sent_at: Mapped[datetime | None] = mapped_column(DateTime)

    created_at: Mapped[datetime] = mapped_column(DateTime, default=utcnow)
    updated_at: Mapped[datetime] = mapped_column(DateTime, default=utcnow, onupdate=utcnow)

    marked_by = relationship("User", foreign_keys=[marked_by_id])

    # --- Просрочка вычисляется, а не хранится: исправили paid_at → пересчиталась ---

    def overdue_days(self, on: date) -> int:
        """Сколько дней просрочки. Оплата в день срока — вовремя (0)."""
        if self.status == STATUS_PAID and self.paid_at:
            return max((self.paid_at - self.due_date).days, 0)
        return max((on - self.due_date).days, 0)

    def state(self, on: date) -> str:
        if self.status == STATUS_PAID:
            return STATE_PAID_LATE if self.overdue_days(on) > 0 else STATE_PAID
        return STATE_OVERDUE if on > self.due_date else STATE_PENDING

    def __repr__(self) -> str:
        return f"<PaymentOccurrence {self.id} rp={self.recurring_payment_id} {self.due_date} {self.status}>"

from app.auth import DB, CurrentUser
from app.errors import ApiError

from app.models.payment_occurrence import (
    STATE_OVERDUE,
    STATE_PAID,
    STATE_PAID_LATE,
    STATE_PENDING,
    STATUS_PAID,
    STATUS_UNPAID,
)

StateFilter = Literal["pending", "overdue", "paid", "paid_late", "unpaid_all"]


def _state_filter(state: str, on: date):
    O = PaymentOccurrence
    if state == STATE_PENDING:
        return and_(O.status == STATUS_UNPAID, O.due_date >= on)
    if state == STATE_OVERDUE:
        return and_(O.status == STATUS_UNPAID, O.due_date < on)
    if state == STATE_PAID:
        return and_(O.status == STATUS_PAID, or_(O.paid_at.is_(None), O.paid_at <= O.due_date))
    if state == STATE_PAID_LATE:
        return and_(O.status == STATUS_PAID, O.paid_at > O.due_date)
    return O.status == STATUS_UNPAID  # unpaid_all


def _get_occurrence(db: Session, occurrence_id: int) -> PaymentOccurrence:
    occ = db.get(
        PaymentOccurrence,
        occurrence_id,
        options=[joinedload(PaymentOccurrence.marked_by)],
        populate_existing=True,
    )
    if occ is None:
        raise ApiError(404, "Платёж не найден")
    return occ


@router.get(
    "",
    response_model=list[OccurrenceOut],
    summary="Получение списка платежей, оплаченные и просроченные",
)
def list_occurrences(
    db: DB,
    user: CurrentUser,
    state: StateFilter | None = Query(None, description="pending / overdue / paid / paid_late / unpaid_all (= pending + overdue)"),
    status: Status | None = None,
    currency: Currency | None = None,
    recurring_payment_id: int | None = None,
    date_from: date | None = Query(None, description="Срок оплаты с (включительно)"),
    date_to: date | None = Query(None, description="Срок оплаты по (включительно)"),
    include_inactive: bool = Query(True, description="Включать периоды прекращённых платежей"),
    include_future: bool = Query(True, description="Включать ещё не наступившие периоды"),
    order: Literal["asc", "desc"] = "asc",
):
    """Каждый элемент — конкретный период регулярного платежа со своим сроком и статусом.

    Период хранит слепок полей платежа, поэтому список читается по одной таблице.
    """
    on = today()
    ensure_all_occurrences(db, on)

    query = select(PaymentOccurrence).options(joinedload(PaymentOccurrence.marked_by))
    if state:
        query = query.where(_state_filter(state, on))
    if status:
        query = query.where(PaymentOccurrence.status == status)
    if currency:
        query = query.where(PaymentOccurrence.currency == currency)
    if recurring_payment_id:
        query = query.where(PaymentOccurrence.recurring_payment_id == recurring_payment_id)
    if date_from:
        query = query.where(PaymentOccurrence.due_date >= date_from)
    if date_to:
        query = query.where(PaymentOccurrence.due_date <= date_to)
    if not include_inactive:
        query = query.where(PaymentOccurrence.payment_ended_at.is_(None))
    if not include_future:
        query = query.where(PaymentOccurrence.due_date <= on)

    due = PaymentOccurrence.due_date.desc() if order == "desc" else PaymentOccurrence.due_date.asc()
    items = db.scalars(query.order_by(due, PaymentOccurrence.name)).all()
    return [occurrence_to_dict(o, on) for o in items]

    response_model=OccurrenceOut,

)
def get_occurrence(occurrence_id: int, db: DB, user: CurrentUser):

@router.patch(
    "/{occurrence_id}",
    response_model=OccurrenceOut,
    summary="Подтверждение или отмена оплаты, исправление фактической даты оплаты",
    responses={
        404: {"description": "Платёж не найден"},
        409: {"description": "Нельзя исправить дату у неоплаченного периода"},
    },
)
def update_occurrence(occurrence_id: int, data: OccurrenceUpdate, db: DB, user: CurrentUser):
    """
    - `{"status": "paid"}` — оплачено сегодня;
    - `{"status": "paid", "paid_at": "2026-10-20"}` — оплачено в указанную дату;
    - `{"paid_at": "2026-10-19"}` — исправить дату у оплаченного периода
      (просрочка пересчитывается, срок не меняется);
    - `{"status": "unpaid"}` — отменить отметку, дата очищается.

    Доступно и администратору, и пользователю.
    """
    fields = data.model_fields_set
    occ = _get_occurrence(db, occurrence_id)
    on = today()

    if "status" not in fields and occ.status != STATUS_PAID:
        raise ApiError(409, "Сначала отметьте оплату (status=paid), потом исправляйте дату")

    new_status = data.status if "status" in fields and data.status else occ.status
    if new_status == STATUS_PAID:
        if "paid_at" in fields and data.paid_at is None:
            raise ApiError(422, "Для оплаченного периода нужна дата оплаты")
        occ.status = STATUS_PAID
        occ.paid_at = data.paid_at or occ.paid_at or on
        occ.marked_by_id = user.id
    else:
        occ.status = STATUS_UNPAID
        occ.paid_at = None
        occ.marked_by_id = None

    db.flush()
    # Оплатили текущий период заранее → сразу появится следующий
    payment = db.get(RecurringPayment, occ.recurring_payment_id)
    if payment is not None:
        ensure_occurrences(db, payment, on)
    db.commit()
    return occurrence_to_dict(_get_occurrence(db, occurrence_id), on)

    Оплаченные и уже наступившие периоды остаются в истории;
    будущие неоплаченные периоды удаляются."""

        raise ApiError(404, "Платёж не найден или уже прекращён")

            PaymentOccurrence.recurring_payment_id == rp.id,
            PaymentOccurrence.status == STATUS_UNPAID,

        .where(PaymentOccurrence.recurring_payment_id == rp.id)
        .values(payment_ended_at=rp.ended_at)
    )
    db.commit()
    return Response(status_code=204)

router = APIRouter(prefix="/payments", tags=["summary"])


@router.get("/summary", response_model=SummaryOut, summary="Получение итоговых сумм платежей по валютам")
def summary(
    db: DB,
    user: CurrentUser,
    date_from: date | None = Query(None, description="Срок оплаты с (включительно)"),
    date_to: date | None = Query(None, description="Срок оплаты по (включительно)"),
    on_date: date | None = Query(None, description="Дата, на которую считаются статусы и просрочка (по умолчанию сегодня)"),
):
    """Суммы считаются отдельно по каждой валюте (RUB, USD, EUR), конвертации нет."""
    on = on_date or today()
    ensure_all_occurrences(db, today())

    query = select(PaymentOccurrence)
    if date_from:
        query = query.where(PaymentOccurrence.due_date >= date_from)
    if date_to:
        query = query.where(PaymentOccurrence.due_date <= date_to)

    zero = Decimal("0.00")
    totals = {
        cur: {
            "currency": cur,
            "total": zero, "paid": zero, "unpaid": zero, "overdue": zero,
            "count_total": 0, "count_paid": 0, "count_unpaid": 0, "count_overdue": 0,
        }
        for cur in CURRENCIES
    }

    for occ in db.scalars(query):
        t = totals[occ.currency]
        t["total"] += occ.amount
        t["count_total"] += 1
        # Статус «на дату»: оплата после on_date считается ещё не сделанной
        paid_on_date = occ.status == STATUS_PAID and (occ.paid_at is None or occ.paid_at <= on)
        if paid_on_date:
            t["paid"] += occ.amount
            t["count_paid"] += 1
        else:
            t["unpaid"] += occ.amount
            t["count_unpaid"] += 1
            if on > occ.due_date:
                t["overdue"] += occ.amount
                t["count_overdue"] += 1

    return {"date_from": date_from, "date_to": date_to, "on_date": on, "currencies": list(totals.values())}

from decimal import Decimal, InvalidOperation
from typing import Annotated

from pydantic import BaseModel, BeforeValidator, ConfigDict, Field, PlainSerializer, WithJsonSchema


def _money_in(value) -> Decimal:
    """Принимаем число или строку ("1500", "1 500,50") — на вход."""
    if value is None:
        raise ValueError("Сумма обязательна")
    try:
        amount = Decimal(str(value).strip().replace(" ", "").replace(",", "."))
    except (InvalidOperation, ValueError):
        raise ValueError("Сумма должна быть числом, например 1500.00")
    if amount <= 0:
        raise ValueError("Сумма должна быть больше нуля")
    return amount


# Вход: платёж нулём не бывает — сумма обязательна и больше нуля.
MoneyIn = Annotated[
    Decimal,
    BeforeValidator(_money_in),
    Field(max_digits=14, decimal_places=2),
]

# Выход: в ответе сумма — любое неотрицательное число (в итогах бывает 0.00),
# поэтому проверку «больше нуля» применяем только ко входным данным.
MoneyOut = Annotated[
    Decimal,
    Field(max_digits=14, decimal_places=2),
    PlainSerializer(lambda v: f"{Decimal(v):.2f}", return_type=str),
    WithJsonSchema({"type": "string", "example": "50000.00", "description": "Сумма, до 2 знаков после запятой"}),
]


def _utc_iso(value: datetime) -> str:
    """Метки времени храним в БД как naive UTC; наружу отдаём с явным +00:00."""
    if value.tzinfo is None:
        value = value.replace(tzinfo=timezone.utc)
    return value.isoformat()


UtcDateTime = Annotated[
    datetime,
    PlainSerializer(_utc_iso, return_type=str),
    WithJsonSchema({"type": "string", "format": "date-time"}),
]


class Schema(BaseModel):
    model_config = ConfigDict(from_attributes=True)


class MessageOut(Schema):
    message: str

"""Генерация периодов (PaymentOccurrence) — ЗАГЛУШКА этапа «каркас».

Контракт зафиксирован, реализация — следующий этап. Что здесь должно быть
(правила из кейса):

- у каждого периода своя дата и своя отметка «Оплачено / Не оплачено»;
- новый период не стирает прошлые оплаты;
- просрочка не останавливает следующие периоды;
- нет такого числа в месяце → последний день месяца (см. app.timeutils.add_months);
- период создаётся только для активного платежа.

Период — самостоятельная запись: name, amount и currency копируются из шаблона
(слепок), payment_ended_at = None. Благодаря этому список периодов читается
по одной таблице, без JOIN с recurring_payments.

Пока реализация не готова, функции ничего не создают: списки платежей
приходят пустыми, а тесты на периоды помечены как ожидающие
(tests/test_domain_logic.py).
"""

from datetime import date

from sqlalchemy import select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from app.logging_config import get_logger
from app.models import PaymentOccurrence, RecurringPayment
from app.models.payment_occurrence import STATUS_UNPAID
from app.timeutils import add_months, today

logger = get_logger(__name__)

# Предохранитель от бесконечного цикла (100 лет помесячно)
MAX_PERIODS = 1200


def due_dates(rp: RecurringPayment):
    """Генератор дат оплаты: первая дата и далее с шагом периода."""
    for k in range(MAX_PERIODS):
        yield add_months(rp.first_due_date, k * rp.months_step, rp.anchor_day)


def ensure_occurrences(
    db: Session, rp: RecurringPayment, on: date | None = None
) -> list[PaymentOccurrence]:
    """Досоздаёт недостающие периоды платежа.

    Идём от первой даты и останавливаемся на первом неоплаченном периоде
    со сроком сегодня или позже: прошлые периоды уже созданы и оплачены,
    текущий — ждёт оплаты, а следующий появится, как только текущий закроют.
    """
    if not rp.is_active:
        return []

    on = on or today()
    if rp.id is None:
        db.flush()

    existing = {
        o.due_date: o
        for o in db.scalars(
            select(PaymentOccurrence).where(
                PaymentOccurrence.recurring_payment_id == rp.id
            )
        )
    }

    created = []

    for due in due_dates(rp):
        occ = existing.get(due)
        if occ is None:
            occ = _create_occurrence(db, rp, due)
            if occ is not None:
                created.append(occ)
            else:
                # Параллельно создал другой процесс — забираем его запись
                occ = db.scalars(
                    select(PaymentOccurrence).where(
                        PaymentOccurrence.recurring_payment_id == rp.id,
                        PaymentOccurrence.due_date == due,
                    )
                ).one()
            existing[due] = occ
        if due >= on and occ.status == STATUS_UNPAID:
            break
    if created:
        logger.info(
            f"Платёж #{rp.id} «{rp.name}»: создано периодов {len(created)} "
            f"({created[0].due_date} … {created[-1].due_date})"
        )
    return created

def _create_occurrence(db: Session, rp: RecurringPayment, due: date) -> PaymentOccurrence | None:
    """Создаёт период в savepoint: дубль (unique-constraint) не ломает транзакцию."""
    occ = PaymentOccurrence(
        recurring_payment_id=rp.id,
        name=rp.name,
        amount=rp.amount,
        currency=rp.currency,
        due_date=due,
        status=STATUS_UNPAID,
    )
    try:
        with db.begin_nested():
            db.add(occ)
        return occ
    except IntegrityError:
        logger.warning(f"Период {due} для платежа #{rp.id} уже создан другим процессом")
        return None

def ensure_all_occurrences(db: Session, on: date | None = None) -> int:
    """Досоздаёт периоды для всех активных платежей. Возвращает число новых."""
    total = 0
    for rp in db.scalars(select(RecurringPayment).where(RecurringPayment.is_active.is_(True))):
        total += len(ensure_occurrences(db, rp, on))
    if total:
        db.commit()
    return total

"""Напоминания об оплате в Telegram.

Правила из кейса:
- минимум за REMINDER_DAYS_BEFORE (3) дня до срока;
- только неоплаченные периоды действующих платежей;
- без дублей: период «захватывается» атомарным UPDATE ... WHERE reminder_sent_at IS NULL,
  поэтому повторная проверка или параллельный процесс второй раз не отправят;
- добавили платёж позже, чем за 3 дня до срока → напоминание сразу (remind_for_payment).

Период хранит слепок полей платежа, поэтому выборка идёт по одной таблице.
"""

import html
from datetime import date, timedelta
from decimal import Decimal

from sqlalchemy import select, update
from sqlalchemy.orm import Session

from app.config import settings
from app.logging_config import get_logger
from app.models import PaymentOccurrence, RecurringPayment
from app.models.payment_occurrence import STATUS_UNPAID
from app.services import telegram
from app.services.periods import ensure_all_occurrences
from app.timeutils import today, utcnow

logger = get_logger(__name__)

# Флаг, чтобы не сыпать предупреждения каждую минуту, пока бот не настроен
_warned_not_configured = False


def format_amount(amount: Decimal, currency: str) -> str:
    whole = f"{Decimal(amount):,.2f}".replace(",", " ")
    return f"{whole} {currency}"


def days_word(n: int) -> str:
    if n % 10 == 1 and n % 100 != 11:
        return "день"
    if n % 10 in (2, 3, 4) and n % 100 not in (12, 13, 14):
        return "дня"
    return "дней"


def build_message(occ: PaymentOccurrence, on: date) -> str:
    """Текст напоминания: название, сумма, валюта и дата оплаты."""
    left = (occ.due_date - on).days
    if left == 0:
        when = "сегодня"
    elif left == 1:
        when = "завтра"
    else:
        when = f"через {left} {days_word(left)}"

    return (
        "🔔 <b>Напоминание об оплате</b>\n\n"
        f"<b>{html.escape(occ.name)}</b>\n"
        f"Сумма: <b>{format_amount(occ.amount, occ.currency)}</b>\n"
        f"Валюта: {occ.currency}\n"
        f"Дата оплаты: <b>{occ.due_date.strftime('%d.%m.%Y')}</b> ({when})"
    )


def _due_occurrences_query(on: date):
    """Неоплаченные периоды действующих платежей со сроком в окне [сегодня; сегодня + N]."""
    window_end = on + timedelta(days=settings.reminder_days_before)
    return (
        select(PaymentOccurrence)
        .where(
            PaymentOccurrence.payment_ended_at.is_(None),   # платёж не прекращён
            PaymentOccurrence.status == STATUS_UNPAID,      # ещё не оплачено
            PaymentOccurrence.reminder_sent_at.is_(None),   # ещё не напоминали
            PaymentOccurrence.due_date >= on,
            PaymentOccurrence.due_date <= window_end,
        )
        .order_by(PaymentOccurrence.due_date)
    )


def _claim(db: Session, occ_id: int) -> bool:
    """Атомарно помечает период как «напоминание отправляется». True — захватили мы."""
    result = db.execute(
        update(PaymentOccurrence)
        .where(PaymentOccurrence.id == occ_id, PaymentOccurrence.reminder_sent_at.is_(None))
        .values(reminder_sent_at=utcnow())
    )
    db.commit()
    return result.rowcount == 1


def _release(db: Session, occ_id: int) -> None:
    """Отправка не удалась — снимаем отметку, чтобы попробовать в следующий раз."""
    db.execute(
        update(PaymentOccurrence)
        .where(PaymentOccurrence.id == occ_id)
        .values(reminder_sent_at=None)
    )
    db.commit()


def _send_for(db: Session, occurrences: list[PaymentOccurrence], on: date) -> int:
    global _warned_not_configured
    if not occurrences:
        return 0
    if not telegram.is_configured():
        if not _warned_not_configured:
            logger.warning(
                "Telegram не настроен (TELEGRAM_BOT_TOKEN / TELEGRAM_CHAT_ID) — "
                "напоминания не отправляются"
            )
            _warned_not_configured = True
        return 0

    sent = 0
    for occ in occurrences:
        # Текст собираем заранее, до захвата: не зависим от состояния объекта после commit
        text = build_message(occ, on)
        occ_id, name, due = occ.id, occ.name, occ.due_date
        if not _claim(db, occ_id):
            logger.info(f"Напоминание по периоду #{occ_id} уже отправлено, пропускаю")
            continue
        try:
            telegram.send_message(text)
            sent += 1
            logger.info(f"Напоминание отправлено: «{name}», срок {due} (период #{occ_id})")
        except Exception:
            logger.exception(f"Не удалось отправить напоминание по периоду #{occ_id}")
            _release(db, occ_id)
    return sent


def check_and_send(db: Session, on: date | None = None) -> int:
    """Полная проверка (вызывается worker'ом по расписанию)."""
    on = on or today()
    # Периоды, которых ещё нет, досоздаём здесь: платёж могли добавить после
    # последнего открытия панели — без этого worker его бы не увидел.
    ensure_all_occurrences(db, on)
    return _send_for(db, list(db.scalars(_due_occurrences_query(on)).all()), on)


def remind_for_payment(db: Session, rp: RecurringPayment, on: date | None = None) -> int:
    """Немедленная проверка одного платежа — сразу после его создания."""
    on = on or today()
    query = _due_occurrences_query(on).where(PaymentOccurrence.recurring_payment_id == rp.id)
    return _send_for(db, list(db.scalars(query).all()), on)

"""Отправка сообщений в Telegram через Bot API.

Кейсу нужен только исходящий поток: бот сам проверяет сроки и пишет в указанный
чат. Обработка команд не требуется, поэтому ходим прямо в Bot API — без aiogram.

Настройка (.env):
    TELEGRAM_BOT_TOKEN=123456:ABC-DEF...   токен от @BotFather
    TELEGRAM_CHAT_ID=-1001234567890        куда писать
"""

import httpx

from app.config import settings
from app.logging_config import get_logger

logger = get_logger(__name__)

API_URL = "https://api.telegram.org/bot{token}/sendMessage"


class TelegramNotConfigured(RuntimeError):
    """В .env нет токена или chat_id."""


def is_configured() -> bool:
    """Настроен ли бот: есть и токен, и чат."""
    return bool(settings.telegram_bot_token and settings.telegram_chat_id)


def send_message(text: str, chat_id: str | None = None) -> dict:
    """Отправляет HTML-сообщение в чат. Бросает исключение при ошибке.

    chat_id можно передать явно — иначе берётся из .env.
    """
    if not is_configured():
        raise TelegramNotConfigured("TELEGRAM_BOT_TOKEN / TELEGRAM_CHAT_ID не заданы в .env")

    response = httpx.post(
        API_URL.format(token=settings.telegram_bot_token),
        json={
            "chat_id": chat_id or settings.telegram_chat_id,
            "text": text,
            "parse_mode": "HTML",
            "disable_web_page_preview": True,
        },
        timeout=10,
    )

    data = response.json() if response.content else {}
    if not data.get("ok"):
        # Токен в лог не пишем: только код и описание ошибки от Telegram
        raise RuntimeError(f"Telegram API {response.status_code}: {data.get('description')}")
    logger.info(f"Сообщение отправлено в чат {chat_id or settings.telegram_chat_id}")
    return data

from app import models  # noqa: F401 — регистрирует все модели в Base.metadata
from app.db import Base, engine

config = context.config
if config.config_file_name is not None:
    fileConfig(config.config_file_name, disable_existing_loggers=False)

target_metadata = Base.metadata


def run_migrations_offline() -> None:
    context.configure(
        url=str(engine.url),
        target_metadata=target_metadata,
        literal_binds=True,
        render_as_batch=True,
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    def skip_empty(context_, revision, directives):
        # Не создаём пустые миграции, если схема не менялась
        if getattr(config.cmd_opts, "autogenerate", False) and directives[0].upgrade_ops.is_empty():
            directives[:] = []
            print("No changes in schema detected.")

    with engine.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata,
            # SQLite не умеет многие ALTER TABLE — batch-режим пересоздаёт таблицу
            render_as_batch=True,
            process_revision_directives=skip_empty,
        )
        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

Revision ID: 0002_user_email
Revises: 0001_initial
Create Date: 2026-10-24

Свежая база: 0001 создаёт таблицы по моделям, колонка email уже есть — пропускаем.

import sqlalchemy as sa
from alembic import op

from app.config import settings

# revision identifiers, used by Alembic.
revision = "0002_user_email"
down_revision = "0001_initial"
branch_labels = None
depends_on = None


def _has_email() -> bool:
    return "email" in {c["name"] for c in sa.inspect(op.get_bind()).get_columns("users")}


def upgrade():
    if _has_email():
        return

    with op.batch_alter_table("users") as batch_op:
        batch_op.add_column(sa.Column("email", sa.String(255), nullable=True))

    bind = op.get_bind()
    bind.execute(
        sa.text("UPDATE users SET email = :email WHERE lower(username) = :username"),
        {
            "email": settings.admin_email.strip().lower(),
            "username": settings.admin_username.strip().lower(),
        },
    )
    bind.execute(
        sa.text("UPDATE users SET email = lower(username) || '@example.com' WHERE email IS NULL")
    )

    with op.batch_alter_table("users") as batch_op:
        batch_op.alter_column("email", existing_type=sa.String(255), nullable=False)
        batch_op.create_unique_constraint("uq_users_email", ["email"])


def downgrade():
    with op.batch_alter_table("users") as batch_op:
        batch_op.drop_constraint("uq_users_email", type_="unique")
        batch_op.drop_column("email")

"""Проверка схемы БД перед стартом приложения.

create_all умеет только СОЗДАВАТЬ таблицы: существующие он не меняет.

«no such column» или «NOT NULL constraint failed».

    python -m scripts.check_schema

Код возврата: 0 — схема совпадает с моделями, 1 — есть расхождения.
Скрипт запускается в entrypoint.sh перед стартом uvicorn, поэтому
схема разошлась — контейнер не поднимется и скажет об этом в логе.
"""

import sys

from app.db import schema_drift


def main() -> int:
    missing, extra = schema_drift()

    if missing:
        print("Схема базы не совпадает с моделями. Не хватает:", file=sys.stderr)
        for item in missing:
            print(f"  - {item}", file=sys.stderr)
        print("", file=sys.stderr)
        print(
            "Приложение упадёт на первой записи. Исправить:\n"
            "    cd backend && python -m scripts.reset_db\n"
            "(данные базы удалятся, демо-данные восстановятся)",
            file=sys.stderr,
        )
        return 1

    if extra:
        print("В базе есть лишние объекты (на работу не влияет):")
        for item in extra:
            print(f"  - {item}")
        print("Убрать: cd backend && python -m scripts.reset_db")
    else:
        print("Схема базы совпадает с моделями")
    return 0


if __name__ == "__main__":
    sys.exit(main())

BACKEND_ROOT = Path(__file__).resolve().parents[1]


def db_path() -> Path | None:
    """Путь к файлу SQLite или None, если база не файловая."""
    if not settings.database_url.startswith("sqlite"):
        return None
    raw = engine.url.database or ""
    if not raw or raw == ":memory:":

    return path if path.is_absolute() else BACKEND_ROOT / path

            print(f"удалён {target.name}")

    Base.metadata.create_all(engine)
    print("схема создана по актуальным моделям")

    from scripts import seed_all

"""Администратор из .env (ADMIN_USERNAME / ADMIN_PASSWORD / ADMIN_EMAIL).

Запускается при каждом старте контейнера: если пароль или email в .env
поменяли — они обновятся. Логин не меняется: он и есть признак администратора.
"""

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.auth import hash_password, verify_password
from app.config import settings
from app.logging_config import get_logger
from app.models import User

logger = get_logger(__name__)


def run(db: Session):
    username = settings.admin_username.strip()
    email = settings.admin_email.strip().lower()
    if not username or not settings.admin_password or not email:
        logger.warning("ADMIN_USERNAME / ADMIN_PASSWORD / ADMIN_EMAIL не заданы — администратор не создан")
        return

    admin = db.scalars(select(User).where(User.username == username)).first()
    if admin is None:
        db.add(User(username=username, email=email, password_hash=hash_password(settings.admin_password)))
        logger.info(f"admin: создан администратор {username}")
    else:
        changed = False
        if not verify_password(settings.admin_password, admin.password_hash):
            admin.password_hash = hash_password(settings.admin_password)
            logger.info(f"admin: пароль администратора {username} обновлён из .env")
            changed = True
        if admin.email != email:
            admin.email = email
            logger.info(f"admin: email администратора {username} обновлён из .env")
            changed = True
        if not changed:
            logger.info(f"admin: аккаунт {username} актуален")

    db.commit()

"""Проверка схемы БД.

create_all умеет только создавать таблицы — существующие он не меняет.
Эти тесты проверяют, что расхождение схемы с моделями находится
до старта приложения (scripts/check_schema.py), а не всплывает позже
на первой записи.
"""

from app import models  # noqa: F401 — регистрирует модели в Base.metadata
from app.db import Base, make_engine, schema_drift


def test_fresh_db_has_no_drift(tmp_path):
    engine = make_engine(f"sqlite:///{tmp_path / 'new.db'}")
    Base.metadata.create_all(engine)

    missing, extra = schema_drift(engine)

    assert missing == []
    assert extra == []


def test_drift_detects_missing_column(tmp_path):
    engine = make_engine(f"sqlite:///{tmp_path / 'old.db'}")
    Base.metadata.create_all(engine)
    with engine.begin() as conn:
        conn.exec_driver_sql("ALTER TABLE users DROP COLUMN updated_at")

    missing, extra = schema_drift(engine)

    assert "users.updated_at" in missing


def test_drift_detects_missing_table(tmp_path):
    engine = make_engine(f"sqlite:///{tmp_path / 'part.db'}")
    Base.metadata.create_all(engine)
    with engine.begin() as conn:
        conn.exec_driver_sql("DROP TABLE revoked_tokens")

    missing, extra = schema_drift(engine)

    assert "таблица revoked_tokens" in missing


def test_extra_column_is_reported_but_not_fatal(tmp_path):
    """Лишняя колонка на работу не влияет — но о ней тоже полезно знать."""
    engine = make_engine(f"sqlite:///{tmp_path / 'extra.db'}")
    Base.metadata.create_all(engine)
    with engine.begin() as conn:
        conn.exec_driver_sql("ALTER TABLE users ADD COLUMN legacy TEXT")

    missing, extra = schema_drift(engine)

    assert missing == []
    assert "users.legacy" in extra
